# Function to get star rating display
def get_star_rating(rating):
    full_stars = int(rating)
    half_star = rating - full_stars >= 0.5
    empty_stars = 5 - full_stars - (1 if half_star else 0)
    
    stars = "★" * full_stars
    if half_star:
        stars += "½"
    stars += "☆" * empty_stars
    return stars
